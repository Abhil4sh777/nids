# Packages the Raspberry Pi needs on top of the project's requirements.txt.
#
# NTLFlowLyzer declares no dependencies in its setup.py, so its two runtime
# dependencies are pinned here instead.  Both are pure-Python wheels, so they
# install on any Python/architecture the bundle targets.
#
# tokenizers gives the ExecuTorch backend exact prompt token counts against
# the shipped tokenizer.json instead of character-based estimates.  Without it
# the system still works (falls back to estimates and logs a warning), but on
# the Pi where the 4k window is tight, exact counts prevent prompt truncation.
#
# This file is the single source for these pins: deploy/make_bundle.py
# downloads exactly these wheels and deploy/install.sh installs exactly these
# wheels.  Do not duplicate the versions anywhere else.

multipledispatch==1.0.0
tokenizers==0.21.1
