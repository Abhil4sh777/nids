# Complete pinned install set for the Raspberry Pi 5 appliance.
#
# Target: Ubuntu desktop 24.04 LTS, Python 3.12 (cp312), linux_aarch64.
# deploy/install.sh installs exactly this file into /opt/nids/venv, so nothing
# is resolved at runtime. tests/test_deploy.py fails if a version drifts.
#
# The first block mirrors the application's direct dependencies in
# requirements.txt. The second block is Pi-only: flow extraction, exact prompt
# token counts, and fetching the model files from Hugging Face.
numpy==2.5.4
pandas==3.1.1
scikit-learn==1.9.1
xgboost==3.4.1
shap==0.54.1
joblib==1.6.0
streamlit==1.67.0
pytest==8.4.2

# NTLFlowLyzer declares no dependencies in its setup.py, so its two runtime
# dependencies are pinned here.  Both are pure-Python wheels.
dpkt==1.9.8
multipledispatch==1.0.0

# tokenizers gives the ExecuTorch backend exact prompt token counts against the
# shipped tokenizer.json instead of character-based estimates.  Without it the
# system still works (falling back to estimates and logging a warning), but on
# the Pi the 4k window is tight, so exact counts prevent prompt truncation.
tokenizers==0.21.1

# huggingface_hub provides the 'hf' command that install.sh uses to fetch the
# ExecuTorch model files when --download-llm is given.  It is not imported by
# the application itself.  Keep it below 1.0 while tokenizers 0.21.1 requires
# huggingface-hub<1.0.
huggingface_hub==0.36.2
